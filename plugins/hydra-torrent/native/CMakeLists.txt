cmake_minimum_required(VERSION 3.20)
project(hydra_torrent_native VERSION 0.1.0 LANGUAGES CXX)
option(HYDRA_STATIC_ENGINE "Build the pinned libtorrent engine into the plugin" ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(static_runtime ON CACHE BOOL "" FORCE)
endif()
if(HYDRA_STATIC_ENGINE)
    include(FetchContent)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    set(build_tests OFF CACHE BOOL "" FORCE)
    set(webtorrent OFF CACHE BOOL "" FORCE)
    set(build_examples OFF CACHE BOOL "" FORCE)
    set(python-bindings OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(libtorrent
        GIT_REPOSITORY https://github.com/arvidn/libtorrent.git
        GIT_TAG 6da363d2994f17c0b3c0450d124cf73a31a73847
        GIT_SUBMODULES deps/try_signal)
    FetchContent_MakeAvailable(libtorrent)
    add_library(LibtorrentRasterbar::torrent-rasterbar ALIAS torrent-rasterbar)
else()
    find_package(LibtorrentRasterbar 2.0 CONFIG REQUIRED)
endif()
find_package(nlohmann_json 3.10 REQUIRED)
add_library(hydra-torrent-native SHARED main.cpp)
set_target_properties(hydra-torrent-native PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
target_include_directories(hydra-torrent-native PRIVATE ../../../crates/hydra-plugin-api/include)
target_compile_features(hydra-torrent-native PRIVATE cxx_std_17)
target_link_libraries(hydra-torrent-native PRIVATE LibtorrentRasterbar::torrent-rasterbar nlohmann_json::nlohmann_json)
if(HYDRA_STATIC_ENGINE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Bind bundled dependencies locally, including OpenSSL ARM64 archive relocations.
    target_link_options(hydra-torrent-native PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    set_property(TARGET hydra-torrent-native APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(MSVC)
    # Keep source locations for native crash dumps without changing release optimization.
    target_compile_options(hydra-torrent-native PRIVATE /Zi)
    if(HYDRA_STATIC_ENGINE)
        target_compile_options(torrent-rasterbar PRIVATE /Zi)
    endif()
    target_link_options(hydra-torrent-native PRIVATE /DEBUG /OPT:REF /OPT:ICF)
    target_compile_options(hydra-torrent-native PRIVATE /W4 /WX)
else()
    target_compile_options(hydra-torrent-native PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS hydra-torrent-native LIBRARY DESTINATION . RUNTIME DESTINATION .)
